local RayParams = RaycastParams.new()
RayParams.FilterType = Enum.RaycastFilterType.Exclude
RayParams.IgnoreWater = true

local function IsVisible(part, origin, ignoreModel)
    local dir = part.Position - origin
    if dir.Magnitude < 0.1 then
        return true
    end

    local filter = {}
    local myChar = LocalPlayer.Character
    if myChar then
        filter[#filter + 1] = myChar
    end
    if ignoreModel then
        filter[#filter + 1] = ignoreModel
    end
    RayParams.FilterDescendantsInstances = filter

    local result = Workspace:Raycast(origin, dir, RayParams)
    if not result then
        return true
    end
    local hit = result.Instance
    return hit == part or hit:IsDescendantOf(part)
end

local function IsGodName(rawName)
    local lower = string.lower(rawName)
    if not string.find(lower, "god", 1, true) then
        return false
    end
    local clean = string.gsub(lower, "[%s_-]", "")
    for i = 1, #CLEAN_GODS do
        if string.find(clean, CLEAN_GODS[i], 1, true) then
            return true
        end
    end
    for i = 1, #FALSE_POSITIVES do
        if string.find(clean, FALSE_POSITIVES[i], 1, true) then
            return false
        end
    end
    return true
end

local function ResolveGodRoot(god)
    if god:IsA("BasePart") then
        return god
    end
    local pp = god.PrimaryPart
    if pp and pp.Parent then
        return pp
    end
    local children = god:GetChildren()
    for i = 1, #children do
        if children[i]:IsA("BasePart") then
            return children[i]
        end
    end
    local descendants = god:GetDescendants()
    for i = 1, #descendants do
        if descendants[i]:IsA("BasePart") then
            return descendants[i]
        end
    end
    return nil
end

local function ComputeGodOffset(god)
    local size = Vector3.new(0, 0, 0)
    if god:IsA("BasePart") then
        size = god.Size
    elseif god:IsA("Model") then
        local ok, s = pcall(function()
            local bbCF, bbSize = god:GetBoundingBox()
            return bbSize
        end)
        if ok and s then
            size = s
        end
    end
    if size.Y > 0 then
        return Vector3.new(0, size.Y * 0.5 + 3, 0)
    end
    return Vector3.new(0, 6, 0)
end

local function UpdateGodVisuals(god, data, cam, now)
    if not god.Parent then
        if data.folder then
            pcall(function() data.folder:Destroy() end)
        end
        GodESPData[god] = nil
        return
    end

    local root = data.root
    if not root or not root.Parent then
        root = nil
        if (now - data.rootRetryAt) >= 1 then
            data.rootRetryAt = now
            root = ResolveGodRoot(god)
            data.root = root
        end
        if not root then
            data.highlight.Enabled = false
            data.billboard.Enabled = false
            return
        end
    end

    local dist = (root.Position - cam.CFrame.Position).Magnitude
    if dist > Settings.GodMaxDist then
        if data.rangeOn then
            data.rangeOn = false
            data.highlight.Enabled = false
            data.billboard.Enabled = false
        end
        return
    end
    data.rangeOn = true

    local visible = true
    if Settings.GodVisCheck then
        visible = IsVisible(root, cam.CFrame.Position, god)
    end

    data.highlight.Enabled = Settings.GodHighlight and visible
    data.highlight.FillColor = Settings.GodColor
    data.highlight.OutlineColor = Settings.GodColor

    if data.lastRoot ~= root then
        data.lastRoot = root
        data.billboard.Adornee = root
    end

    local showText = Settings.GodName or Settings.GodDistance
    data.billboard.Enabled = showText
    if showText then
        local text = ""
        if Settings.GodName then
            text = god.Name
        end
        if Settings.GodDistance then
            if text ~= "" then
                text = text .. " | "
            end
            text = text .. string.format("%.0fm", dist)
        end
        if data.lastText ~= text then
            data.lastText = text
            data.label.Text = text
        end
        if data.labelColor ~= Settings.GodColor then
            data.labelColor = Settings.GodColor
            data.label.TextColor3 = Settings.GodColor
        end
        data.billboard.MaxDistance = Settings.GodMaxDist
    end
end

local function TryRegisterGod(inst)
    if GodESPData[inst] then
        return
    end
    if not (inst:IsA("Model") or inst:IsA("BasePart")) then
        return
    end
    if not IsGodName(inst.Name) then
        return
    end
    if Players:GetPlayerFromCharacter(inst) then
        return
    end
    CreateGodVisuals(inst)
end

local function StartGodScan()
    task.spawn(function()
        local ok, err = pcall(function()
            local stack = {Workspace}
            local processed = 0
            while #stack > 0 and Active and Settings.GodsESP do
                local inst = table.remove(stack)
                TryRegisterGod(inst)
                local children = inst:GetChildren()
                for i = 1, #children do
                    stack[#stack + 1] = children[i]
                end
                processed = processed + 1
                if processed % 4096 == 0 then
                    task.wait()
                end
            end
        end)
        if not ok then
            warn("[ESP] Erro no scan de deuses: " .. tostring(err))
        end
    end)
end

local function HideSkeleton(data)
    if not data.skelOn then
        return
    end
    data.skelOn = false
    local lines = data.skeletonLines
    for i = 1, #lines do
        pcall(function() lines[i].Visible = false end)
    end
end

local function UpdateSkeleton(data, char, cam, color)
    if not Settings.PlayerSkeleton or not DrawingSupported or not char then
        HideSkeleton(data)
        return
    end

    local list = SKELETON_R6
    if not char:FindFirstChild("Torso") then
        if char:FindFirstChild("UpperTorso") then
            list = SKELETON_R15
        else
            list = nil
        end
    end
    if not list then
        HideSkeleton(data)
        return
    end

    data.skelOn = true
    local lines = data.skeletonLines
    for i = 1, #list do
        local line = lines[i]
        if not line then
            local ok, l = pcall(Drawing.new, "Line")
            if ok then
                l.Thickness = 1
                l.Transparency = 1
                l.Visible = false
                lines[i] = l
                line = l
            end
        end
        if line then
            local a = char:FindFirstChild(list[i][1])
            local b = char:FindFirstChild(list[i][2])
            if a and b then
                local pa, oa = cam:WorldToViewportPoint(a.Position)
                local pb, ob = cam:WorldToViewportPoint(b.Position)
                if oa and ob then
                    line.From = Vector2.new(pa.X, pa.Y)
                    line.To = Vector2.new(pb.X, pb.Y)
                    line.Color = color
                    if not line.Visible then
                        line.Visible = true
                    end
                elseif line.Visible then
                    line.Visible = false
                end
            elseif line.Visible then
                line.Visible = false
            end
        end
    end
    for i = #list + 1, #lines do
        if lines[i] and lines[i].Visible then
            lines[i].Visible = false
        end
    end
end

local function UpdatePlayerHeavy(player, data, cam)
    local char = player.Character
    local humanoid = nil
    local root = nil
    local head = nil

    if char then
        humanoid = char:FindFirstChildOfClass("Humanoid")
        root = char:FindFirstChild("HumanoidRootPart")
        if not root then
            root = char.PrimaryPart
        end
        if not root then
            root = char:FindFirstChildWhichIsA("BasePart")
        end
        head = char:FindFirstChild("Head")
        if not head then
            head = root
        end
    end

    data.char = char
    data.root = root
    data.head = head

    local visible = false
    local dist = math.huge
    local hp = 0

    if root and head and humanoid and humanoid.Health > 0 then
        dist = (root.Position - cam.CFrame.Position).Magnitude
        hp = math.clamp(humanoid.Health / math.max(humanoid.MaxHealth, 0.01), 0, 1)
        if dist <= Settings.PlayerMaxDist then
            if Settings.PlayerVisCheck then
                visible = IsVisible(head, cam.CFrame.Position, char)
            else
                visible = true
            end
        end
    end

    data.dist = dist
    data.hp = hp
    data.visible = visible

    if not visible then
        if data.shown then
            data.shown = false
            data.chams.Enabled = false
            data.billboard.Enabled = false
            if data.box3d then
                data.box3d.Visible = false
            end
            data.box2d.Visible = false
            data.healthbar.Visible = false
            data.boxShown = false
            data.hbShown = false
            HideSkeleton(data)
        end
        return
    end
    data.shown = true

    local color = Settings.PlayerColor
    if Settings.PlayerTeamColor and player.Team then
        color = player.TeamColor.Color
    end
    if data.color ~= color then
        data.color = color
        data.box2dStroke.Color = color
        data.chams.FillColor = color
        data.chams.OutlineColor = color
        data.nameLabel.TextColor3 = color
        data.distLabel.TextColor3 = color
        if data.box3d then
            data.box3d.Color3 = color
        end
    end

    if data.lastAdorn ~= char then
        data.lastAdorn = char
        data.chams.Adornee = char
        if data.box3d then
            data.box3d.Adornee = char
        end
        data.billboard.Adornee = head
    end

    data.chams.Enabled = Settings.PlayerChams

    if data.box3d then
        if data.box3d.Parent ~= cam then
            local ok = pcall(function() data.box3d.Parent = cam end)
            if not ok then
                local nb = Instance.new("SelectionBox")
                nb.SurfaceTransparency = 1
                nb.LineThickness = 0.05
                nb.Color3 = color
                nb.Adornee = char
                nb.Visible = false
                nb.Parent = cam
                data.box3d = nb
            end
        end
        data.box3d.Visible = Settings.PlayerBox3D
    end

    local anyText = Settings.PlayerName or Settings.PlayerDistance or Settings.PlayerHealth
    data.billboard.Enabled = anyText
    if anyText then
        data.nameLabel.Visible = Settings.PlayerName
        data.distLabel.Visible = Settings.PlayerDistance
        data.healthLabel.Visible = Settings.PlayerHealth

        if Settings.PlayerName then
            local t = player.Name
            if player.DisplayName ~= player.Name then
                t = player.DisplayName .. " (" .. player.Name .. ")"
            end
            if data.lastName ~= t then
                data.lastName = t
                data.nameLabel.Text = t
            end
        end
        if Settings.PlayerDistance then
            local t = string.format("%.0fm", dist)
            if data.lastDist ~= t then
                data.lastDist = t
                data.distLabel.Text = t
            end
        end
        if Settings.PlayerHealth then
            local t = string.format("%d/%d", math.floor(humanoid.Health + 0.5), math.floor(humanoid.MaxHealth + 0.5))
            if data.lastHpText ~= t then
                data.lastHpText = t
                data.healthLabel.Text = t
            end
        end
    end

    if data.boxShown and not Settings.PlayerBox2D then
        data.boxShown = false
        data.box2d.Visible = false
    end
    if data.hbShown and not Settings.PlayerHealthBar then
        data.hbShown = false
        data.healthbar.Visible = false
    end
    if data.skelOn and (not Settings.PlayerSkeleton or not DrawingSupported) then
        HideSkeleton(data)
    end
end

local function UpdatePlayerFrame(data, cam)
    local char = data.char

    if Settings.PlayerSkeleton and DrawingSupported and char then
        UpdateSkeleton(data, char, cam, data.color or Settings.PlayerColor)
    else
        HideSkeleton(data)
    end

    if not (Settings.PlayerBox2D or Settings.PlayerHealthBar) then
        if data.boxShown then
            data.boxShown = false
            data.box2d.Visible = false
        end
        if data.hbShown then
            data.hbShown = false
            data.healthbar.Visible = false
        end
        return
    end

    local head = data.head
    local root = data.root
    if not head or not root then
        return
    end

    local headPos, headOn = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 1, 0))
    local footPos, footOn = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))

    if headOn and footOn then
        local height = math.abs(footPos.Y - headPos.Y)
        if height < 6 then
            height = 6
        end
        local width = math.max(height * 0.6, 10)
        local x = headPos.X - width * 0.5
        local y = math.min(headPos.Y, footPos.Y)

        if Settings.PlayerBox2D then
            data.box2d.Position = UDim2.fromOffset(x, y)
            data.box2d.Size = UDim2.fromOffset(width, height)
            if not data.boxShown then
                data.boxShown = true
                data.box2d.Visible = true
            end
        elseif data.boxShown then
            data.boxShown = false
            data.box2d.Visible = false
        end

        if Settings.PlayerHealthBar then
            data.healthbar.Position = UDim2.fromOffset(x - 8, y)
            data.healthbar.Size = UDim2.fromOffset(4, height)
            if data.lastHp ~= data.hp then
                data.lastHp = data.hp
                data.healthbarFill.Size = UDim2.fromScale(1, data.hp)
                data.healthbarFill.BackgroundColor3 = Color3.new(1 - data.hp, data.hp, 0)
            end
            if not data.hbShown then
                data.hbShown = true
                data.healthbar.Visible = true
            end
        elseif data.hbShown then
            data.hbShown = false
            data.healthbar.Visible = false
        end
    else
        if data.boxShown then
            data.boxShown = false
            data.box2d.Visible = false
        end
        if data.hbShown then
            data.hbShown = false
            data.healthbar.Visible = false
        end
    end
end

local function AutoEquipTool()
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then
        return
    end
    if char:FindFirstChildOfClass("Tool") then
        return
    end
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if not backpack then
        return
    end
    local tools = backpack:GetChildren()
    local best = nil
    for i = 1, #tools do
        local tool = tools[i]
        if tool:IsA("Tool") then
            local n = string.lower(tool.Name)
            if string.find(n, "pick", 1, true) or string.find(n, "axe", 1, true) or string.find(n, "maul", 1, true) then
                best = tool
                break
            end
            if not best then
                best = tool
            end
        end
    end
    if best then
        pcall(function()
            humanoid:EquipTool(best)
        end)
    end
end

local MineBlacklist = {}
local RESOURCE_FOLDER_NAMES = { "Resources", "Resource", "Nodes", "Ores" }

local function GetInstCenter(inst)
    if inst:IsA("BasePart") then
        return inst.Position
    end
    if inst:IsA("Model") then
        local pp = inst.PrimaryPart
        if pp and pp.Parent then
            return pp.Position
        end
        local base = inst:FindFirstChildWhichIsA("BasePart", true)
        if base then
            return base.Position
        end
    end
    return nil
end

local function FindMineTarget(rootPos, now)
    local folders = {}
    for i = 1, #RESOURCE_FOLDER_NAMES do
        local f = Workspace:FindFirstChild(RESOURCE_FOLDER_NAMES[i])
        if f then
            folders[#folders + 1] = f
        end
    end
    if #folders == 0 then
        return nil
    end

    local counts = {}
    local candidates = {}
    for f = 1, #folders do
        local children = folders[f]:GetChildren()
        for i = 1, #children do
            local child = children[i]
            if child:IsA("Model") or child:IsA("BasePart") then
                counts[child.Name] = (counts[child.Name] or 0) + 1
                candidates[#candidates + 1] = child
            end
        end
    end

    local bestGod = nil
    local bestGodDist = Settings.AutoMineMaxDist
    local bestRes = nil
    local bestResDist = Settings.AutoMineMaxDist

    for i = 1, #candidates do
        local cand = candidates[i]
        local bl = MineBlacklist[cand]
        if bl ~= nil and bl < now then
            MineBlacklist[cand] = nil
            bl = nil
        end
        if bl == nil then
            local pos = GetInstCenter(cand)
            if pos then
                local d = (pos - rootPos).Magnitude
                if IsGodName(cand.Name) then
                    if Settings.AutoMineGods and d < bestGodDist then
                        bestGodDist = d
                        bestGod = cand
                    end
                else
                    local nameCount = counts[cand.Name] or 0
                    if nameCount >= Settings.AutoMineMinInstances and d < bestResDist then
                        bestResDist = d
                        bestRes = cand
                    end
                end
            end
        end
    end

    if bestGod then
        if Settings.AutoMinePreferGods or not bestRes or bestGodDist <= bestResDist then
            return bestGod
        end
    end
    return bestRes
end

--============================--
-- AUTO MINERAR - Spy do Remote (ByteNetReliable)
--============================--
local RemoteSpyData = {
    remote = nil,
    original = nil,
    hooked = false,
    armed = false,
    captured = {},
}

local function FindByteNetRemote()
    local rs = game:GetService("ReplicatedStorage")
    local found = rs:FindFirstChild("ByteNetReliable", true)
    if not found then
        found = Workspace:FindFirstChild("ByteNetReliable", true)
    end
    if found then
        RemoteSpyData.remote = found
        return found
    end
    return nil
end

local function PackArgs(...)
    local t = { ... }
    t.n = select("#", ...)
    return t
end

local function InstallRemoteSpy()
    if RemoteSpyData.hooked then
        return true
    end
    if type(hookmetamethod) ~= "function" or type(getnamecallmethod) ~= "function" then
        return false
    end
    local remote = RemoteSpyData.remote or FindByteNetRemote()
    if not remote then
        return false
    end

    local oldNamecall
    local hookFunc = function(self, ...)
        local method = getnamecallmethod()
        if method == "FireServer" and self == remote then
            local args = PackArgs(...)
            if RemoteSpyData.armed then
                pcall(function()
                    local idx = #RemoteSpyData.captured + 1
                    RemoteSpyData.captured[idx] = args
                    print("[AutoMine] Pacote " .. idx .. " capturado (" .. tostring(args.n) .. " argumentos)")
                end)
            end
        end
        return oldNamecall(self, ...)
    end

    if type(newcclosure) == "function" then
        pcall(function()
            hookFunc = newcclosure(hookFunc)
        end)
    end

    local ok = pcall(function()
        oldNamecall = hookmetamethod(game, "__namecall", hookFunc)
    end)
    if ok then
        RemoteSpyData.hooked = true
        RemoteSpyData.original = oldNamecall
        return true
    end
    return false
end

local function UninstallRemoteSpy()
    if not RemoteSpyData.hooked then
        return
    end
    RemoteSpyData.hooked = false
    RemoteSpyData.armed = false
    pcall(function()
        if RemoteSpyData.original and type(hookmetamethod) == "function" then
            hookmetamethod(game, "__namecall", RemoteSpyData.original)
        end
    end)
end

local function RemoteMineStep()
    local remote = RemoteSpyData.remote or FindByteNetRemote()
    if not remote then
        return false
    end
    local list = RemoteSpyData.captured
    local count = #list
    if count == 0 then
        return false
    end
    local idx = Settings.AutoMinePacketIndex
    if idx < 1 or idx > count then
        idx = count
    end
    local args = list[idx]
    local ok = pcall(function()
        remote:FireServer(table.unpack(args, 1, args.n))
    end)
    return ok
end

--============================--
-- AUTO MINERAR - Ciclo principal
--============================--
local AutoMineRunning = false

local function AutoMineCycle()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        return 1
    end

    local now = os.clock()
    local target = FindMineTarget(root.Position, now)
    if not target then
        return 1
    end

    local targetPos = GetInstCenter(target)
    if not targetPos then
        MineBlacklist[target] = os.clock() + 3
        return 0.5
    end

    if Settings.AutoMineEquipTool then
        AutoEquipTool()
    end
    if Settings.AutoMineStopMove then
        ReleaseMovementKeys()
    end

    AimPos = targetPos
    BindAim()
    task.wait(0.2)

    local cam = Workspace.CurrentCamera
    if not cam then
        AimPos = nil
        UnbindAim()
        return 0.5
    end

    local sp = cam:WorldToViewportPoint(targetPos)
    local cx = sp.X
    local cy = sp.Y
    local vw = cam.ViewportSize.X
    local vh = cam.ViewportSize.Y
    if cx < 2 then
        cx = 2
    elseif cx > vw - 2 then
        cx = vw - 2
    end
    if cy < 2 then
        cy = 2
    elseif cy > vh - 2 then
        cy = vh - 2
    end

    if Settings.GodDebug then
        print("[AutoMine] Minerando: " .. target.Name .. " (" .. string.format("%.0fm", (targetPos - root.Position).Magnitude) .. ")")
    end

    MineInputDown(cx, cy)

    local deadline = os.clock() + Settings.AutoMineHoldTime
    while AutoMineRunning and Active and os.clock() < deadline do
        if not target.Parent then
            break
        end
        MineInputUpdate(cx, cy)
        task.wait(0.15)
    end

    MineInputUp(cx, cy)
    local mined = not target.Parent
    if not mined then
        MineBlacklist[target] = os.clock() + Settings.AutoMineRetryDelay
    end

    AimPos = nil

    if mined then
        return 0.25
    end
    return 0.6
end

local function StartAutoMine()
    if AutoMineRunning then
        return
    end
    AutoMineRunning = true

    if Settings.AutoMineMethod == "Remote" then
        local okSpy = InstallRemoteSpy()
        if not okSpy then
            print("[AutoMine] Modo Remote indisponivel (sem hookmetamethod ou remote nao achado). Usando Input.")
            Settings.AutoMineMethod = "Input"
        end
    end

    task.spawn(function()
        local hintShown = false
        while AutoMineRunning and Active do
            local waitTime = 1
            local ok, err = pcall(function()
                if Settings.AutoMineMethod == "Remote" then
                    if #RemoteSpyData.captured > 0 then
                        RemoteMineStep()
                        waitTime = Settings.AutoMineRemoteDelay
                    else
                        if not hintShown then
                            hintShown = true
                            print("[AutoMine] Modo Remote: nenhum pacote capturado. Use o botao de captura e mine 1x manualmente.")
                        end
                        waitTime = 1
                    end
                else
                    waitTime = AutoMineCycle()
                end
            end)
            if not ok then
                warn("[AutoMine] Erro: " .. tostring(err))
                waitTime = 1
            end
            task.wait(waitTime)
        end
        MineInputUp(LastPressX, LastPressY)
        AimPos = nil
        UnbindAim()
        AutoMineRunning = false
    end)
end

local function StopAutoMine()
    AutoMineRunning = false
    AimPos = nil
    UnbindAim()
    MineInputUp(LastPressX, LastPressY)
    table.clear(MineBlacklist)
end

local MiscTab = Window:CreateTab({ Name = "Misc", Icon = 4779510615 })

MiscTab:CreateButton({
    Name = "Limpar beacons antigos (GodBeacon)",
    Callback = function()
        task.spawn(function()
            local n = 0
            local descendants = Workspace:GetDescendants()
            for i = 1, #descendants do
                if descendants[i].Name == "GodBeacon" then
                    pcall(function() descendants[i]:Destroy() end)
                    n = n + 1
                end
            end
            print("[ESP] GodBeacons removidos: " .. n)
        end)
    end,
})

MiscTab:CreateButton({
    Name = "Destruir este script (limpa tudo)",
    Callback = function()
        FullCleanup()
    end,
})

})
